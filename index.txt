    <script>
      const video = document.querySelector('#ar-video-src');
      const heart = document.querySelector('#ar-heart');
      const btn = document.querySelector('#action-btn');

      // Клик по кнопке запускает видео со звуком
      btn.addEventListener('click', () => {
        video.play();
        btn.style.display = 'none';
      });

      // Когда видео закончится, прячем его и показываем 3D-сердце
      video.addEventListener('ended', () => {
        document.querySelector('#ar-video').setAttribute('visible', 'false');
        heart.setAttribute('visible', 'true');
      });
